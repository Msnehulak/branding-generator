from pathlib import Path
import os
from xml.etree import ElementTree as ET
from svgpathtools import svg2paths, Path as SvgPath
import cairosvg

BASE_DIR = Path(__file__).resolve().parent


RESOLUTIONS = {
    'sd': (640, 480),
    'hd': (1280, 720),
    'fhd': (1920, 1080),
    '4k': (3840, 2160),
}

FOLDERS = {
    'row': BASE_DIR / 'base',
    'crop_row': BASE_DIR / 'crop',
    'brand': BASE_DIR / 'branding',
}

class Branding:
    def __init__(self) -> None:
        for folder in FOLDERS.values():
            folder.mkdir(parents=True, exist_ok=True)

    @staticmethod
    def _crop_svg(input: str | Path, output: str | Path) -> None:
        try:
            paths, _ = svg2paths(str(input))
            if not paths:
                return

            combined = SvgPath(*paths)
            xmin, xmax, ymin, ymax = combined.bbox()
            width = xmax - xmin
            height = ymax - ymin

            if width <= 0 or height <= 0:
                return

            tree = ET.parse(str(input))
            root = tree.getroot()

            root.set("viewBox", f"{xmin} {ymin} {width} {height}")
            root.set("width", f"{width}px")
            root.set("height", f"{height}px")

            tree.write(output, encoding="utf-8", xml_declaration=True)
        except Exception as e:
            print(f"❌ Chyba při ořezu {input}: {e}")

    @staticmethod
    def _save_as_png(
        input_path: str | Path,
        output_path: str | Path,
        size: str = 'hd',
        transparent: bool = True,
    ) -> None:
        try:
            width, height = RESOLUTIONS.get(size.lower(), (1280, 720))
            
            kwargs = {
                "url": str(input_path),
                "write_to": str(output_path),
                "output_width": width,
                "output_height": height,
            }
            
            if not transparent:
                kwargs["background_color"] = "white"

            cairosvg.svg2png(**kwargs)
            print(f"✅ PNG úspěšně vyexportováno do {output_path}")
        except Exception as e:
            print(f"❌ Chyba při exportu PNG {input_path}: {e}")

    @staticmethod
    def files_in_path(path, ends = '.svg'):
        file = []
        for f in os.listdir(path):
            f = str(f)
            if f.endswith(ends):
                file.append(f)
        return file

    def crop_svgs(self):
        row = self.files_in_path(FOLDERS['row'])
        print('--- START: Croping')
        for f in row:
            inp = BASE_DIR / FOLDERS['row'] / f
            out = BASE_DIR / FOLDERS['crop_row'] / f 
            self._crop_svg(inp, out)
            print(f'>>> Crop {f}')
        print('--- END: Croping')
        
    def main(self):
        self.crop_svgs()

if __name__ == "__main__":
    app = Branding()
    app.main()
